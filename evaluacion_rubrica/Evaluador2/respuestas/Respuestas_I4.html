FUNCTION_BLOCK FB_EstrellaTriangulo
VAR_INPUT
    btnMarcha : BOOL; // Botón de marcha
    btnParo : BOOL;   // Botón de paro
END_VAR

VAR_OUTPUT
    kM : BOOL;       // Contactor principal
    kY : BOOL;       // Contactor de estrella
    kD : BOOL;       // Contactor de triángulo
END_VAR

VAR
    timerEstrella : TON; // Temporizador para el contacto de estrella
    timerMuerto : TON;   // Temporizador para el tiempo muerto
    estado : INT;       // Variable de estado del arranque
END_VAR

// Inicialización de temporizadores
timerEstrella(IN := FALSE, PT := T#8S);
timerMuerto(IN := FALSE, PT := T#100MS);

// Máquina de estados
CASE estado OF
    0: // Estado inicial
        IF btnMarcha THEN
            kM := TRUE; // Activar el contactor principal
            kY := TRUE; // Activar el contactor de estrella
            timerEstrella(IN := TRUE); // Iniciar temporizador para el contacto de estrella
            estado := 1; // Transición al estado 1
        END_IF

    1: // Esperando el fin del temporizador de estrella
        IF timerEstrella.Q THEN
            kY := FALSE; // Desactivar el contactor de estrella
            timerMuerto(IN := TRUE); // Iniciar temporizador para el tiempo muerto
            estado := 2; // Transición al estado 2
        END_IF

    2: // Esperando el fin del temporizador de tiempo muerto
        IF timerMuerto.Q THEN
            kD := TRUE; // Activar el contactor de triángulo
            estado := 3; // Transición al estado 3
        END_IF

    3: // Estado final
        // Mantener los contactores activados según el estado actual
        IF btnParo THEN
            kM := FALSE; // Desactivar el contactor principal
            kY := FALSE; // Desactivar el contactor de estrella
            kD := FALSE; // Desactivar el contactor de triángulo
            estado := 0; // Reiniciar a estado inicial
        END_IF
END_CASE

VAR_INPUT
    btnMarcha : BOOL; (* Boton de marcha *)
    btnParo   : BOOL; (* Boton de paro *)
END_VAR
VAR_OUTPUT
    kM : BOOL; (* Contactor principal *)
    kY : BOOL; (* Contactor estrella *)
    kD : BOOL; (* Contactor triangulo *)
END_VAR
VAR
    marcha    : BOOL; (* Memoria de marcha (enclavamiento) *)
    tEstrella : TON;  (* Temporizador fase estrella: 8 s *)
    tMuerto   : TON;  (* Temporizador tiempo muerto: 100 ms *)
END_VAR

(* --- Gestion del enclavamiento de marcha/paro --- *)
IF btnParo THEN
    marcha := FALSE; (* Paro inmediato *)
ELSIF btnMarcha THEN
    marcha := TRUE;  (* Arranque *)
END_IF;

(* --- Temporizador de fase estrella (8 s desde el arranque) --- *)
tEstrella(IN := marcha, PT := T#8S);

(* --- Temporizador de tiempo muerto (100 ms tras fin de estrella) --- *)
tMuerto(IN := tEstrella.Q, PT := T#100MS);

(* --- Asignacion de salidas --- *)
kM := marcha;                        (* Contactor principal activo durante toda la marcha *)
kY := marcha AND NOT tEstrella.Q;    (* Estrella: activo hasta que vence el temporizador de 8 s *)
kD := marcha AND tMuerto.Q;          (* Triangulo: activo solo tras el tiempo muerto *)

(* Nota de seguridad: kY y kD no pueden ser TRUE simultaneamente porque
   kY se desactiva cuando tEstrella.Q=TRUE y kD se activa solo cuando
   tMuerto.Q=TRUE (despues de los 100 ms adicionales). *)
END_FUNCTION_BLOCK

FUNCTION_BLOCK FB_EstrellaTriangulo
VAR_INPUT
    btnMarcha : BOOL; (* Botón de inicio del motor *)
    btnParo   : BOOL; (* Botón de parada del motor *)
END_VAR

VAR_OUTPUT
    kM : BOOL; (* Contactor principal *)
    kY : BOOL; (* Contactor de estrella *)
    kD : BOOL; (* Contactor de triángulo *)
END_VAR

VAR
    xEnMarcha   : BOOL; (* Memoria de estado del motor *)
    tonEstrella : TON;  (* Temporizador para la fase de estrella *)
    tonTrans    : TON;  (* Temporizador para el tiempo muerto (transición) *)
END_VAR

(* Lógica de retención (Marcha / Paro) *)
IF btnParo THEN
    xEnMarcha := FALSE;
ELSIF btnMarcha THEN
    xEnMarcha := TRUE;
END_IF;

(* Temporizador de la fase Estrella (8 segundos) *)
tonEstrella(IN := xEnMarcha, PT := T#8S);

(* Temporizador de tiempo muerto entre Estrella y Triángulo (100 ms) *)
(* Inicia cuando termina la fase de estrella *)
tonTrans(IN := tonEstrella.Q, PT := T#100MS);

(* Asignación de salidas con enclavamientos de seguridad *)
(* El contactor principal se activa siempre que el sistema esté en marcha *)
kM := xEnMarcha;

(* Estrella: Activo en marcha, antes de los 8s, y NUNCA si Triángulo está activo *)
kY := xEnMarcha AND NOT tonEstrella.Q AND NOT kD;

(* Triángulo: Activo en marcha, tras el tiempo muerto, y NUNCA si Estrella está activo *)
kD := xEnMarcha AND tonTrans.Q AND NOT kY;

END_FUNCTION_BLOCK

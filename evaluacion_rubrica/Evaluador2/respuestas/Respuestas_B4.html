    La version anterior daba veredicto 1, y sin levantar la bandera, a una
    implementacion con una CARRERA DE INTERLOCK: el enclavamiento cruzado se
    evaluaba contra el valor de la otra salida en el scan ANTERIOR. Con el
    motor ya girando se comporta bien, pero al pulsar las dos ordenes ESTANDO
    EN REPOSO ambas salidas se activan a la vez durante un scan.

    Causa: la secuencia solo pulsaba las dos ordenes con el motor ya girando en
    reversa (paso 6), situacion en la que la salida activa bloquea a la otra
    incluso con la comparacion retrasada. Nunca las pulsaba desde el reposo,
    que es justamente el caso en que la carrera se manifiesta.

    Es un fallo de COBERTURA, no del mecanismo: la enmienda 4 dejo el
    instrumento capaz de detectar la violacion transitoria, pero la secuencia
    no llegaba a provocarla. Una bandera enclavada solo puede atrapar lo que la
    secuencia produce.

    Correccion: se anaden dos pasos que pulsan ambas ordenes desde el reposo.
    El paso 8 no exige un sentido concreto —el enunciado no dice cual debe
    ganar si se piden los dos a la vez— sino unicamente que no coexistan.

FUNCTION_BLOCK FB_Giro
VAR_INPUT
    btnFwd : BOOL;  // Botón para girar en sentido horario
    btnRev : BOOL;  // Botón para girar en sentido antihorario
    btnParo : BOOL; // Botón para parar el motor
END_VAR

VAR_OUTPUT
    fwd : BOOL;  // Salida para girar en sentido horario
    rev : BOOL;  // Salida para girar en sentido antihorario
END_VAR

VAR
    estado : INT := 0; // Variable de estado: 0 - Parado, 1 - Giro horario, 2 - Giro antihorario
END_VAR

// Máquina de estados
CASE estado OF
    0: // Estado Parado
        fwd := FALSE;
        rev := FALSE;
        IF btnFwd THEN
            estado := 1; // Cambia a giro horario
        ELSIF btnRev THEN
            estado := 2; // Cambia a giro antihorario
        END_IF

    1: // Estado Giro Horario
        fwd := TRUE;
        rev := FALSE;
        IF btnParo THEN
            estado := 0; // Cambia a parado
        END_IF

    2: // Estado Giro Antihorario
        fwd := FALSE;
        rev := TRUE;
        IF btnParo THEN
            estado := 0; // Cambia a parado
        END_IF
END_CASE

FUNCTION_BLOCK FB_Giro
VAR_INPUT
    btnFwd : BOOL; (* Boton para giro horario *)
    btnRev : BOOL; (* Boton para giro antihorario *)
    btnParo : BOOL; (* Boton de paro *)
END_VAR

    fwd : BOOL; (* Salida para giro horario *)
    rev : BOOL; (* Salida para giro antihorario *)

    estado : INT; (* Estado del motor: 0 = Paro, 1 = Horario, 2 = Antihorario *)
END_VAR

(* Logica de control para inversion de giro del motor *)
IF btnParo THEN
    estado := 0; (* Parar el motor *)
ELSIF btnFwd AND (estado = 0) THEN
    estado := 1; (* Activar giro horario si el motor esta parado *)
ELSIF btnRev AND (estado = 0) THEN
    estado := 2; (* Activar giro antihorario si el motor esta parado *)
END_IF

(* Actualizar salidas segun el estado *)
fwd := (estado = 1);
rev := (estado = 2);
END_FUNCTION_BLOCK

FUNCTION_BLOCK FB_Giro
VAR_INPUT
    btnFwd  : BOOL; (* Botón marcha horaria *)
    btnRev  : BOOL; (* Botón marcha antihoraria *)
    btnParo : BOOL; (* Botón paro *)
END_VAR

VAR_OUTPUT
    fwd : BOOL; (* Salida giro horario *)
    rev : BOOL; (* Salida giro antihorario *)
END_VAR

VAR
    (* Estado interno: TRUE = en marcha, FALSE = parado *)
    xRunning : BOOL := FALSE;
END_VAR

(*
    Lógica de control:
    - btnParo detiene el motor (prioridad máxima).
    - btnFwd activa fwd solo si el motor está parado (rev = FALSE).
    - btnRev activa rev solo si el motor está parado (fwd = FALSE).
    - Nunca se permite cambio directo de sentido sin pasar por paro.
*)

(* ── 1. PARO: prioridad máxima ── *)
IF btnParo THEN
    fwd      := FALSE;
    rev      := FALSE;
    xRunning := FALSE;

(* ── 2. MARCHA HORARIA: solo si el motor está parado ── *)
ELSIF btnFwd AND NOT xRunning THEN
    fwd      := TRUE;
    rev      := FALSE;  (* Enclavamiento: rev siempre apagada *)
    xRunning := TRUE;

(* ── 3. MARCHA ANTIHORARIA: solo si el motor está parado ── *)
ELSIF btnRev AND NOT xRunning THEN
    rev      := TRUE;
    fwd      := FALSE;  (* Enclavamiento: fwd siempre apagada *)
    xRunning := TRUE;

(* ── 4. Sin pulsación relevante: mantener estado actual ── *)
END_IF;

(* ── 5. Enclavamiento de seguridad por hardware (doble seguro) ── *)
(* Garantiza que nunca ambas salidas estén activas, incluso ante  *)
(* corrupción de memoria o fallo lógico imprevisto.               *)
IF fwd AND rev THEN
    fwd      := FALSE;
    rev      := FALSE;
    xRunning := FALSE;
END_IF;

END_FUNCTION_BLOCK

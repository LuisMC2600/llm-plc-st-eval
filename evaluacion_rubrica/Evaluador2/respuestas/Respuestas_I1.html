FUNCTION_BLOCK FB_Semaforo
VAR_INPUT
    habilitado : BOOL; (* Habilita la secuencia normal del semáforo *)
END_VAR

VAR_OUTPUT
    luzV : BOOL; (* Luz verde *)
    luzA : BOOL; (* Luz amarilla *)
    luzR : BOOL; (* Luz roja *)
END_VAR

VAR
    (* Temporizadores para cada fase de la secuencia normal *)
    tVerde    : TON; (* Temporiza la fase verde  - 20 s *)
    tAmarillo : TON; (* Temporiza la fase amarilla - 4 s *)
    tRojo     : TON; (* Temporiza la fase roja   - 15 s *)

    (* Temporizador para el amarillo intermitente *)
    tIntermitente : TON; (* Temporiza cada semiciclo de 1 s *)

    (* Estado interno de la secuencia normal *)
    (* 0 = Verde, 1 = Amarillo, 2 = Rojo *)
    estado : INT := 0;

    (* Bit de parpadeo para el amarillo intermitente *)
    parpadeo : BOOL := FALSE;
END_VAR

(* ─────────────────────────────────────────────────────────────
   Lógica principal
   ───────────────────────────────────────────────────────────── *)

IF NOT habilitado THEN
    (* ── Modo amarillo intermitente ─────────────────────────── *)

    (* Reiniciar temporizadores de secuencia normal *)
    tVerde(IN    := FALSE, PT := T#20S);
    tAmarillo(IN := FALSE, PT := T#4S);
    tRojo(IN     := FALSE, PT := T#15S);

    (* Reiniciar estado para que al habilitar arranque en verde *)
    estado := 0;

    (* Temporizador de 1 s para el parpadeo *)
    tIntermitente(IN := TRUE, PT := T#1S);

    IF tIntermitente.Q THEN
        (* Conmutar el bit de parpadeo y reiniciar el temporizador *)
        parpadeo          := NOT parpadeo;
        tIntermitente(IN  := FALSE, PT := T#1S); (* Reset *)
        tIntermitente(IN  := TRUE,  PT := T#1S); (* Rearranque *)
    END_IF;

    (* Salidas: solo amarillo parpadea *)
    luzV := FALSE;
    luzA := parpadeo;
    luzR := FALSE;

ELSE
    (* ── Modo secuencia normal ───────────────────────────────── *)

    (* Detener y resetear el temporizador de intermitente *)
    tIntermitente(IN := FALSE, PT := T#1S);
    parpadeo := FALSE;

    CASE estado OF

        0: (* ── Verde ── *)
            luzV := TRUE;
            luzA := FALSE;
            luzR := FALSE;

            tVerde(IN := TRUE, PT := T#20S);

            IF tVerde.Q THEN
                tVerde(IN := FALSE, PT := T#20S); (* Reset *)
                estado := 1;                       (* Pasar a amarillo *)
            END_IF;

        1: (* ── Amarillo ── *)
            luzV := FALSE;
            luzA := TRUE;
            luzR := FALSE;

            tAmarillo(IN := TRUE, PT := T#4S);

            IF tAmarillo.Q THEN
                tAmarillo(IN := FALSE, PT := T#4S); (* Reset *)
                estado := 2;                         (* Pasar a rojo *)
            END_IF;

        2: (* ── Rojo ── *)
            luzV := FALSE;
            luzA := FALSE;
            luzR := TRUE;

            tRojo(IN := TRUE, PT := T#15S);

            IF tRojo.Q THEN
                tRojo(IN := FALSE, PT := T#15S); (* Reset *)
                estado := 0;                      (* Volver a verde *)
            END_IF;

        ELSE
            (* Estado inválido: reiniciar en verde *)
            estado := 0;

    END_CASE;

END_IF;

FUNCTION_BLOCK FB_Semaforo
VAR_INPUT
    habilitado : BOOL; (* Habilita la secuencia normal *)
END_VAR
VAR_OUTPUT
    luzV : BOOL; (* Luz verde *)
    luzA : BOOL; (* Luz amarilla *)
    luzR : BOOL; (* Luz roja *)
END_VAR
VAR
    estado : INT := 0; (* 0: Verde, 1: Amarillo, 2: Rojo *)
    tVerde : TON;      (* Temporizador luz verde *)
    tAmarillo : TON;   (* Temporizador luz amarilla *)
    tRojo : TON;       (* Temporizador luz roja *)
    tBlink1 : TON;     (* Temporizador intermitencia ON *)
    tBlink2 : TON;     (* Temporizador intermitencia OFF *)
END_VAR

(* Control de estados y reinicio *)
IF NOT habilitado THEN
    estado := 0; (* Reinicia la secuencia a verde si se deshabilita *)
ELSIF estado = 0 AND tVerde.Q THEN
    estado := 1;
ELSIF estado = 1 AND tAmarillo.Q THEN
    estado := 2;
ELSIF estado = 2 AND tRojo.Q THEN
    estado := 0;
END_IF;

(* Temporizadores de la secuencia normal *)
tVerde(IN := habilitado AND (estado = 0), PT := T#20S);
tAmarillo(IN := habilitado AND (estado = 1), PT := T#4S);
tRojo(IN := habilitado AND (estado = 2), PT := T#15S);

(* Temporizadores del modo intermitente (1s ON / 1s OFF) *)
tBlink1(IN := NOT habilitado AND NOT tBlink2.Q, PT := T#1S);
tBlink2(IN := tBlink1.Q, PT := T#1S);

(* Asignacion de salidas *)
luzV := habilitado AND (estado = 0);
luzR := habilitado AND (estado = 2);
luzA := (habilitado AND (estado = 1)) OR (NOT habilitado AND NOT tBlink1.Q);

END_FUNCTION_BLOCK

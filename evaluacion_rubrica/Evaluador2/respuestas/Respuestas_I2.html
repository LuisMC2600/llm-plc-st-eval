    durante la mezcla obtenia el mismo resultado que la correcta. El paro en

    bloque realmente mezclando. Se anade ademas un paso final que comprueba que
    tras el paro el proceso permanece en reposo y no rearranca solo.

VAR_INPUT

    nivelAlto : BOOL;   // Sensor de nivel alto
    nivelBajo : BOOL;   // Sensor de nivel bajo

    vLlenado : BOOL;    // Válvula de llenado
    motorMezcla : BOOL; // Motor del mezclador
    vDescarga : BOOL;   // Válvula de descarga

VAR
    estado : INT := 0;  // Máquina de estados (0:Reposo, 1:Llenado, 2:Mezcla, 3:Descarga)
    timerMezcla : TON;  // Temporizador para la etapa de mezcla
END_VAR

// 1. Condición de paro prioritario
IF btnParo THEN
    estado := 0;
END_IF;

// 2. Máquina de estados secuencial
CASE estado OF
    0: // Etapa 0: Reposo
        IF btnInicio AND NOT btnParo THEN
            estado := 1;
        END_IF;
        
    1: // Etapa 1: Llenado
        IF nivelAlto THEN
            estado := 2;
        END_IF;
        
    2: // Etapa 2: Mezcla

            estado := 3;
        END_IF;
        
    3: // Etapa 3: Descarga
        IF nivelBajo THEN
            estado := 0;
        END_IF;
ELSE
    // Protección contra estados inválidos
    estado := 0; 
END_CASE;

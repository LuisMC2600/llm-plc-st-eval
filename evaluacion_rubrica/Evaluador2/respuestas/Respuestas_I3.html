    la alarma AUNQUE la causa siguiera presente, que es la violacion directa de

    ACK tras retirarla (si debe)". El paso estaba aprobado en el documento pero
    no implementado en el banco.

    Correccion: se anade el paso 2, que aplica ack3 con alm3 todavia activa y
    exige que la alarma NO se borre. La secuencia pasa de 9 a 10 pasos.

FUNCTION_BLOCK FB_Alarmas
VAR_INPUT
    alm1 : BOOL; // Alarma de prioridad alta
    alm2 : BOOL; // Alarma de prioridad media
    alm3 : BOOL; // Alarma de prioridad baja
    ack1 : BOOL; // Reconocimiento de alarma 1
    ack2 : BOOL; // Reconocimiento de alarma 2
    ack3 : BOOL; // Reconocimiento de alarma 3
END_VAR

VAR_OUTPUT
    sirena : BOOL; // Indica si la sirena está activa
    codigoAlm : INT; // Código de la alarma de mayor prioridad activa
END_VAR

VAR
    ret1 : BOOL; // Retención de la alarma 1
    ret2 : BOOL; // Retención de la alarma 2
    ret3 : BOOL; // Retención de la alarma 3
END_VAR

// Retención de alarmas
IF alm1 THEN
    ret1 := TRUE;
END_IF;

IF alm2 THEN
    ret2 := TRUE;
END_IF;

IF alm3 THEN
    ret3 := TRUE;
END_IF;

// Reconocimiento de alarmas
IF ack1 AND NOT alm1 THEN
    ret1 := FALSE;
END_IF;

IF ack2 AND NOT alm2 THEN
    ret2 := FALSE;
END_IF;

IF ack3 AND NOT alm3 THEN
    ret3 := FALSE;
END_IF;

// Determinación de la alarma de mayor prioridad
IF ret1 THEN
    codigoAlm := 1;
ELSIF ret2 THEN
    codigoAlm := 2;
ELSIF ret3 THEN
    codigoAlm := 3;
ELSE
    codigoAlm := 0;
END_IF;

// Activación de la sirena si hay alguna alarma retenida
sirena := ret1 OR ret2 OR ret3;

END_FUNCTION_BLOCK

FUNCTION_BLOCK FB_Alarmas
VAR_INPUT
    alm1 : BOOL; // Causa alarma 1 (Prioridad Alta)
    alm2 : BOOL; // Causa alarma 2 (Prioridad Media)
    alm3 : BOOL; // Causa alarma 3 (Prioridad Baja)
    ack1 : BOOL; // Reconocimiento alarma 1
    ack2 : BOOL; // Reconocimiento alarma 2
    ack3 : BOOL; // Reconocimiento alarma 3
END_VAR

VAR_OUTPUT
    sirena : BOOL;   // Sirena activa si hay alguna alarma presente
    codigoAlm : INT; // Código de la alarma de mayor prioridad (1, 2, 3 o 0 si no hay)
END_VAR

VAR
    estadoAlm1 : BOOL; // Estado retenido de la alarma 1
    estadoAlm2 : BOOL; // Estado retenido de la alarma 2
    estadoAlm3 : BOOL; // Estado retenido de la alarma 3
END_VAR

// 1. Lógica de retención (latch) y reconocimiento
// La alarma se activa si la causa está presente. 
// Se mantiene activa mientras no se reconozca (ack).
// Si la causa sigue presente, el ack no tiene efecto (almX domina).
estadoAlm1 := alm1 OR (estadoAlm1 AND NOT ack1);
estadoAlm2 := alm2 OR (estadoAlm2 AND NOT ack2);
estadoAlm3 := alm3 OR (estadoAlm3 AND NOT ack3);

// 2. Lógica de prioridad para el código de alarma
// Se evalúa en orden descendente de prioridad (1 -> 2 -> 3)
IF estadoAlm1 THEN
    codigoAlm := 1;
ELSIF estadoAlm2 THEN
    codigoAlm := 2;
ELSIF estadoAlm3 THEN
    codigoAlm := 3;
ELSE
    codigoAlm := 0;
END_IF;

// 3. Lógica de activación de la sirena
// Se enciende si cualquiera de los estados retenidos es verdadero
sirena := estadoAlm1 OR estadoAlm2 OR estadoAlm3;

END_FUNCTION_BLOCK

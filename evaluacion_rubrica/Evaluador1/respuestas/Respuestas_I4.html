FUNCTION_BLOCK FB_EstrellaTriangulo
VAR_INPUT
    btnMarcha : BOOL; (* Pulsador de marcha *)
    btnParo   : BOOL; (* Pulsador de paro   *)
END_VAR

VAR_OUTPUT
    kM : BOOL; (* Contactor principal  *)
    kY : BOOL; (* Contactor estrella   *)
    kD : BOOL; (* Contactor triángulo  *)
END_VAR

VAR
    (* Temporizador estrella: 8 s en estrella *)
    tEstrella  : TON;
    (* Temporizador tiempo muerto: 100 ms entre kY y kD *)
    tMuerto    : TON;

    (* Máquina de estados *)
    eEstado    : INT; (* 0=Reposo 1=Estrella 2=TiempoMuerto 3=Triangulo *)
END_VAR

(* ─────────────────────────────────────────────
   Lógica principal - Arranque Estrella-Triángulo
   ───────────────────────────────────────────── *)

(* PARO: tiene prioridad absoluta en cualquier estado *)
IF btnParo THEN
    eEstado := 0;
END_IF;

CASE eEstado OF

    (* ── Estado 0: Reposo ── *)
    0:
        kM := FALSE;
        kY := FALSE;
        kD := FALSE;
        tEstrella(IN := FALSE, PT := T#8S);    (* Resetear temporizadores *)
        tMuerto  (IN := FALSE, PT := T#100MS);

        IF btnMarcha AND NOT btnParo THEN
            eEstado := 1; (* Pasar a estrella *)
        END_IF;

    (* ── Estado 1: Funcionamiento en Estrella ── *)
    1:
        kM := TRUE;
        kY := TRUE;  (* Estrella activa  *)
        kD := FALSE; (* Triángulo inactivo - enclavamiento *)

        tEstrella(IN := TRUE, PT := T#8S);
        tMuerto  (IN := FALSE, PT := T#100MS); (* Mantener reseteado *)

        IF tEstrella.Q THEN
            kY      := FALSE;  (* Desactivar estrella antes de cambiar estado *)
            eEstado := 2;      (* Pasar a tiempo muerto *)
        END_IF;

    (* ── Estado 2: Tiempo Muerto (kY y kD ambos inactivos) ── *)
    2:
        kM := TRUE;
        kY := FALSE; (* Estrella inactiva  - enclavamiento garantizado *)
        kD := FALSE; (* Triángulo inactivo - enclavamiento garantizado *)

        tEstrella(IN := TRUE,  PT := T#8S);    (* Mantener Q activo para no resetear *)
        tMuerto  (IN := TRUE,  PT := T#100MS); (* Iniciar cuenta de tiempo muerto *)

        IF tMuerto.Q THEN
            eEstado := 3; (* Pasar a triángulo *)
        END_IF;

    (* ── Estado 3: Funcionamiento en Triángulo ── *)
    3:
        kM := TRUE;
        kY := FALSE; (* Estrella inactiva  - enclavamiento *)
        kD := TRUE;  (* Triángulo activo   *)

        tEstrella(IN := TRUE, PT := T#8S);    (* Mantener temporizadores estables *)
        tMuerto  (IN := TRUE, PT := T#100MS);

        (* Permanece en triángulo hasta que se pulse btnParo *)

    ELSE
        (* Estado desconocido: ir a reposo de forma segura *)
        eEstado := 0;

END_CASE;

(* Enclavamiento hardware por software: garantía adicional *)
(* kY y kD nunca pueden ser TRUE simultáneamente          *)
IF kY AND kD THEN
    kY := FALSE;
    kD := FALSE;
    kM := FALSE;
    eEstado := 0; (* Fallo: volver a reposo *)
END_IF;

FUNCTION_BLOCK FB_EstrellaTriangulo
VAR
    // Entradas
    btnMarcha : BOOL;
    btnParo : BOOL;
    
    // Salidas
    kM : BOOL := FALSE;
    kY : BOOL := FALSE;
    kD : BOOL := FALSE;
    
    // Variables internas
    estado : INT := 0;
    timerEstrella : TON; // Temporizador para el modo estrella
    timerMuerto : TON;   // Temporizador para el tiempo muerto
END_VAR

// Inicialización de los temporizadores
timerEstrella(IN := FALSE, PT := T#8S);
timerMuerto(IN := FALSE, PT := T#100MS);

// Lógica de control
IF btnParo THEN
    // Si se presiona el botón de paro, desactivar todos los contactores
    kM := FALSE;
    kY := FALSE;
    kD := FALSE;
    estado := 0;
ELSIF btnMarcha THEN
    CASE estado OF
        0: // Estado inicial
            kM := TRUE; // Activar contactor principal
            kY := TRUE; // Activar contactor estrella
            timerEstrella(IN := TRUE); // Iniciar temporizador estrella
            estado := 1;
        
        1: // Esperando que termine el temporizador estrella
            IF timerEstrella.Q THEN
                kY := FALSE; // Desactivar contactor estrella
                timerEstrella(IN := FALSE); // Reiniciar temporizador estrella
                timerMuerto(IN := TRUE); // Iniciar temporizador muerto
                estado := 2;
            END_IF
        
        2: // Esperando que termine el tiempo muerto
            IF timerMuerto.Q THEN
                kD := TRUE; // Activar contactor triángulo
                timerMuerto(IN := FALSE); // Reiniciar temporizador muerto
                estado := 3;
            END_IF
        
        3: // Estado final, contactor triángulo activado
            // Mantener el estado hasta que se presione btnParo
            // o se desactive btnMarcha
            IF NOT btnMarcha THEN
                kM := FALSE;
                kD := FALSE;
                estado := 0;
            END_IF
    END_CASE
ELSE
    // Si btnMarcha no está activo, asegurarse de que todo esté apagado
    kM := FALSE;
    kY := FALSE;
    kD := FALSE;
    estado := 0;
END_IF
END_FUNCTION_BLOCK

FUNCTION_BLOCK FB_EstrellaTriangulo
VAR_INPUT
    btnMarcha : BOOL; // Botón de arranque
    btnParo   : BOOL; // Botón de parada (TRUE = detener)
END_VAR

VAR_OUTPUT
    kM : BOOL; // Contactor principal
    kY : BOOL; // Contactor de estrella
    kD : BOOL; // Contactor de triángulo
END_VAR

VAR
    bRun        : BOOL; // Memoria del estado de marcha
    tonStar     : TON;  // Temporizador para la fase de estrella
    tonDeadTime : TON;  // Temporizador para el tiempo muerto
END_VAR

// 1. Lógica de retención (Marcha / Paro)
// El paro tiene prioridad sobre la marcha
IF btnParo THEN
    bRun := FALSE;
ELSIF btnMarcha THEN
    bRun := TRUE;
END_IF;

// 2. Temporizador de la fase de estrella (8 segundos)
tonStar(IN := bRun, PT := T#8S);

// 3. Control del contactor de Estrella (kY)
// Activo si hay marcha, no ha terminado el tiempo de estrella y kD está apagado (enclavamiento)
kY := bRun AND NOT tonStar.Q AND NOT kD;

// 4. Temporizador de tiempo muerto (100 milisegundos)
// Comienza a contar solo cuando termina la fase de estrella y kY se ha desactivado
tonDeadTime(IN := tonStar.Q AND NOT kY, PT := T#100MS);

// 5. Control del contactor de Triángulo (kD)
// Activo si hay marcha, terminó el tiempo muerto y kY está apagado (enclavamiento)
kD := bRun AND tonDeadTime.Q AND NOT kY;

// 6. Control del contactor Principal (kM)
// Activo siempre que el sistema esté en marcha
kM := bRun;

END_FUNCTION_BLOCK

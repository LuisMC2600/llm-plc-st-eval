VAR_INPUT
    puertaAbierta : BOOL; (* Señal de puerta abierta *)
    reset : BOOL;         (* Señal de reset *)
END_VAR
VAR_OUTPUT
    aviso : BOOL;         (* Señal de aviso *)
    nAperturas : INT;     (* Número de aperturas *)
END_VAR
VAR
    puertaAbiertaAnt : BOOL; (* Estado anterior de la puerta *)
END_VAR

(* Si reset está activo, reiniciar el contador y desactivar el aviso *)
IF reset THEN
    nAperturas := 0;
    aviso := FALSE;
ELSE
    (* Detectar transición de FALSE a TRUE en puertaAbierta *)
    IF NOT puertaAbiertaAnt AND puertaAbierta THEN
        nAperturas := nAperturas + 1;
    END_IF;
    
    (* Activar aviso si el número de aperturas es 5 o más *)
    IF nAperturas >= 5 THEN
        aviso := TRUE;
    ELSE
        aviso := FALSE;
    END_IF;
END_IF;

(* Actualizar el estado anterior de la puerta *)
puertaAbiertaAnt := puertaAbierta;
END_FUNCTION_BLOCK

FUNCTION_BLOCK FB_Puerta
VAR_INPUT
    puertaAbierta : BOOL; (* Señal de puerta: TRUE = abierta *)
    reset         : BOOL; (* Reset: pone contador a 0 y desactiva aviso *)
END_VAR

VAR_OUTPUT
    aviso      : BOOL; (* TRUE cuando nAperturas >= 5 *)
    nAperturas : INT;  (* Número de aperturas registradas *)
END_VAR

VAR
    puertaAbierta_prev : BOOL; (* Valor de puertaAbierta en el ciclo anterior *)
END_VAR

(* ------------------------------------------------------------------ *)
(* Lógica principal                                                     *)
(* ------------------------------------------------------------------ *)

(* Reset tiene prioridad sobre cualquier otra acción *)
IF reset THEN
    nAperturas         := 0;
    aviso              := FALSE;
    puertaAbierta_prev := FALSE; (* Evita flanco espurio tras el reset *)

ELSE
    (* Detección de flanco ascendente: FALSE -> TRUE *)
    IF puertaAbierta AND NOT puertaAbierta_prev THEN
        nAperturas := nAperturas + 1;
    END_IF;

    (* Activar aviso al alcanzar o superar 5 aperturas *)
    IF nAperturas >= 5 THEN
        aviso := TRUE;
    END_IF;

END_IF;

(* Guardar estado actual para el próximo ciclo *)
puertaAbierta_prev := puertaAbierta;

END_FUNCTION_BLOCK

    aviso      : BOOL; (* Aviso al acumular 5 o mas aperturas *)
    nAperturas : INT;  (* Contador de aperturas *)
END_VAR

    rTrigPuerta : R_TRIG; (* Detector de flanco ascendente de la puerta *)
END_VAR

IF reset THEN
    (* Reset: pone el contador en cero y desactiva el aviso *)
    nAperturas := 0;
ELSIF rTrigPuerta.Q THEN
    (* Nueva apertura detectada: incrementar contador *)
    nAperturas := nAperturas + 1;
END_IF;

(* Activar aviso si se han acumulado 5 o mas aperturas *)
aviso := nAperturas >= 5;
END_FUNCTION_BLOCK

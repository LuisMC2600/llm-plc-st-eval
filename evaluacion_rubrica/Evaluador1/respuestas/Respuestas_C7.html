       El resultado era un accidente del entorno, no una medicion.

    con independencia del ciclo de tarea. El temporizador de asentamiento del
    paso queda retenido mientras la generacion esta en curso, de modo que la
    evaluacion solo ocurre cuando ya se emitieron todos los flancos.

    3. (detectado por la validacion) Comprobar solo el estado final (cuenta=12,
       loteListo=TRUE) NO distingue un preset incorrecto: una implementacion que
       active el lote a las 10 piezas tambien llega a 12 con el lote activo. El
       umbral se verifica ahora en DOS LOTES: primero 10 flancos (cuenta=11, el
       lote debe seguir APAGADO) y despues 1 flanco mas (cuenta=12, lote ENCENDIDO).

FUNCTION_BLOCK FB_Contador
VAR_INPUT
    sensorPieza : BOOL; (* Contacto NA para contar *)
    btnReset : BOOL; (* Contacto NA para resetear el contador *)
END_VAR
VAR_OUTPUT
    loteListo : BOOL; (* Bobina que se activa cuando se alcanza el lote *)
    cuenta : INT; (* Valor actual del contador *)
END_VAR
VAR
    C1 : CTU; (* Contador ascendente *)
END_VAR

(* Configuración del contador *)
C1.PV := 12; (* Valor preestablecido del contador *)

(* Rama 1: Conteo *)
IF sensorPieza THEN
    C1.CU := TRUE; (* Incrementa el contador *)
ELSE
    C1.CU := FALSE;
END_IF

(* Rama 2: Reset *)
IF btnReset THEN
    C1.RESET := TRUE; (* Resetea el contador *)
ELSE
    C1.RESET := FALSE;
END_IF

(* Rama 3: Actualización del contador *)
C1(); (* Ejecuta el contador *)
cuenta := C1.CV; (* Transfiere el valor actual del contador a la variable cuenta *)

(* Rama 4: Lote listo *)
loteListo := C1.Q; (* La bobina se activa cuando se alcanza el valor preestablecido *)
END_FUNCTION_BLOCK

FUNCTION_BLOCK FB_Contador
VAR_INPUT
    sensorPieza : BOOL; (* Sensor de detección de pieza *)
    btnReset    : BOOL; (* Botón de reset del contador *)
END_VAR

VAR_OUTPUT
    loteListo : BOOL; (* Indica que el lote de 12 piezas está completo *)
    cuenta    : INT;  (* Valor actual del contador *)
END_VAR

VAR
    C1 : CTU; (* Contador ascendente IEC 61131-3 *)
END_VAR

(* -------------------------------------------------------
   Rama 1: Contador ascendente C1
   CU  <- sensorPieza (flanco ascendente cuenta una pieza)
   RESET <- btnReset  (reinicia el contador)
   PV  = 12           (tamaño del lote)
------------------------------------------------------- *)
C1(
    CU    := sensorPieza,
    RESET := btnReset,
    PV    := 12
);

(* -------------------------------------------------------
   Rama 2: Activar loteListo cuando el contador alcanza PV
   Equivale al contacto normalmente abierto de C1.Q
   alimentando la bobina loteListo
------------------------------------------------------- *)
loteListo := C1.Q;

(* -------------------------------------------------------
   Rama 3: Transferir el valor actual del contador
   a la variable de salida cuenta
------------------------------------------------------- *)
cuenta := C1.CV;

END_FUNCTION_BLOCK
